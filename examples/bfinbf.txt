var MAX_SIZE = 250;
var EXIT_CHAR = '!';  // Exit entering code when hit this character

array cells[250];     // BF cells
var cellPointer = 0;  // Cell pointer
array code[250];      // User entered code
var i = 0;            // Code pointer

// Get user input code
while(i != MAX_SIZE) {
    var char = input;
    code[i] = char;
    i = i + 1;
    if(char == EXIT_CHAR) {
        i = MAX_SIZE;
    }
}

i = 0;
var c = 0; // current code character
var backtrack = 0;

while(i != MAX_SIZE) {
    c = code[i];
    if(c == '+') {
        cells[cellPointer] = cells[cellPointer] + 1;
    } else if(c == '-') {
        cells[cellPointer] = cells[cellPointer] - 1;
    } else if(c == '.') {
        print cells[cellPointer];
    } else if(c == ',') {
        cells[cellPointer] = input;
    } else if(c == '<') {
        cellPointer = cellPointer - 1;
    } else if(c == '>') {
        cellPointer = cellPointer + 1;
    } else if(c == '[') {
        if(!cells[cellPointer]) {
            i = i + 1;
            while(backtrack or code[i] != ']') {
                if(code[i] == '[') { backtrack = backtrack - 1; }
                else if(code[i] == ']') { backtrack = backtrack + 1; }
                i = i + 1;
            }
        }
    } else if(c == ']') {
        if(cells[cellPointer]) {
            i = i - 1;
            while(backtrack or code[i] != '[') {
                if(code[i] == '[') { backtrack = backtrack - 1; }
                else if(code[i] == ']') { backtrack = backtrack + 1; }
                i = i - 1;
            }
            i = i - 1;
        }
    }
    i = i + 1;
}
