var MAX_SIZE = 250;
var EXIT_CHAR = 33;  // Exit entering code when hit this character (!)

array cells[250];     // BF cells
var cellPointer = 0;  // Cell pointer
array code[250];      // User entered code
var i = 0;            // Code pointer

// Get user input code
while(i != MAX_SIZE) {
    var char = input;
    code[i] = char;
    i = i + 1;
    if(char == EXIT_CHAR) {
        i = MAX_SIZE;
    }
}

i = 0;
var c = 0; // current code character
var backtrack = 0;

while(i != MAX_SIZE) {
    c = code[i];
    if(c == 43) {
        // +
        cells[cellPointer] = cells[cellPointer] + 1;
    } else if(c == 45) {
        // -
        cells[cellPointer] = cells[cellPointer] - 1;
    } else if(c == 46) {
        // .
        print cells[cellPointer];
    } else if(c == 44) {
        // ,
        cells[cellPointer] = input;
    } else if(c == 60) {
        // <
        cellPointer = cellPointer - 1;
    } else if(c == 62) {
        // >
        cellPointer = cellPointer + 1;
    } else if(c == 91) {
        // [
        if(!cells[cellPointer]) {
            i = i + 1;
            while((backtrack != 0) or (code[i] != 93)) {
                if(code[i] == 91) { backtrack = backtrack - 1; }
                else if(code[i] == 93) { backtrack = backtrack + 1; }
                i = i + 1;
            }
        }
    } else if(c == 93) {
        // ]
        if(cells[cellPointer]) {
            i = i - 1;
            while((backtrack != 0) or (code[i] != 91)) {
                if(code[i] == 91) { backtrack = backtrack - 1; }
                else if(code[i] == 93) { backtrack = backtrack + 1; }
                i = i - 1;
            }
            i = i - 1;
        }
    }
    i = i + 1;
}
